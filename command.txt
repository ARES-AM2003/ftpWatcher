# Docker build, push, run and inspect commands for ares2003/ftpworker:latest

# 1) Build image locally
docker build -t ares2003/ftpworker:latest .

# 3) Push to Docker Hub (ensure you're logged in)
docker push ares2003/ftpworker:latest
docker pull ares2003/ftpworker:latest
docker stop ftpworker
docker rm ftpworker
# 4) Run container (background, privileged, mounts and restart policy)
docker run -d \
  --name ftpworker \
  --privileged \
  --restart unless-stopped \
  -v /opt/ftp:/opt/ftp \
  -v /opt/ftp/.env:/app/.env \
  -v /srv/ftp:/srv/ftp \
  ares2003/ftpworker:latest
